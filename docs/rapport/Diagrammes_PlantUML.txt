# 3. Diagramme de séquence GÉNÉRIQUE d'une requête (Figure III.3)
#    Cheminement MVC : du navigateur jusqu'à la vue
################################################################################

@startuml Sequence_Generique
title Flux générique d'une requête (architecture MVC)

actor Utilisateur as U
participant "Navigateur" as B
participant "index.php\n(Front Controller)" as F
participant "Routeur" as R
participant "Controller" as C
participant "Model" as M
database "MySQL\n(PDO)" as DB
participant "Vue" as V

U -> B : Action (clic / formulaire)
B -> F : Requête HTTP (GET/POST)
F -> F : bootstrap (autoload, session, aides)
F -> R : dispatch()
R -> R : Recherche route (méthode + chemin)
alt route trouvée
  R -> C : appel méthode du contrôleur
  C -> C : Validation des données
  C -> M : Appel(s) métier
  M -> DB : Requête SQL préparée
  DB --> M : Résultats
  M --> C : Données (tableaux)
  C -> V : Rendu avec les données
  V --> B : Page HTML
  B --> U : Affichage
else route inconnue / méthode invalide
  R --> B : Erreur 404 / 405
  B --> U : Page d'erreur
end
@enduml

# 4. Diagramme de séquence — SOUMISSION D'UN AVIS (Figure III.4)
#    Cas concret : POST /avis
################################################################################

@startuml Sequence_Soumission_Avis
title Soumission d'un avis (POST /avis)

actor Utilisateur as U
participant "Navigateur" as B
participant "Routeur" as R
participant "AvisController" as C
participant "Session" as S
participant "AvisModel" as M
database "MySQL" as DB

U -> B : Remplit et envoie le formulaire d'avis
B -> R : POST /avis
R -> C : traiterSoumission()

C -> C : Vérifie que la requête est un POST
C -> S : estConnecte() ?
alt non connecté
  S --> C : false
  C -> S : flashErreurs("Connexion requise")
  C --> B : Redirection /connexion
else connecté
  S --> C : true
  C -> C : Récupère et valide les champs\n(note 1-5, commentaire <= 8000, lieu)
  alt données invalides
    C -> S : flashErreurs(...)
    C --> B : Redirection page précédente
  else données valides
    C -> M : lieuExiste(id_lieu)
    M -> DB : SELECT 1 FROM lieu WHERE id_lieu=?
    DB --> M : existe
    M --> C : true
    C -> M : utilisateurADejaAvisSurLieu(uid, id_lieu)
    M -> DB : SELECT COUNT(*) FROM avis ...
    DB --> M : 0
    M --> C : false (pas de doublon)
    C -> M : creerPourLieu(uid, id_lieu, note, ...)
    M -> DB : INSERT INTO avis (...) VALUES (...)
    DB --> M : id inséré
    M --> C : id_avis
    C -> S : flashSucces("Avis enregistré")
    C --> B : Redirection /lieu?id=X
    B --> U : Fiche du lieu avec le nouvel avis
  end
end
@enduml
